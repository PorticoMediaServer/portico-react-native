cmake_minimum_required(VERSION 3.22.1)
project(porticomedia C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
foreach(name avcodec avformat avutil swresample ass)
  add_library(${name} SHARED IMPORTED)
  set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${PORTICO_MEDIA_ENGINE}/lib/${ANDROID_ABI}/lib${name}.so")
endforeach()
if(NOT EXISTS "${PORTICO_MEDIA_ENGINE}/include/libavcodec/avcodec.h" OR NOT EXISTS "${PORTICO_MEDIA_ENGINE}/include/ass/ass.h")
  message(FATAL_ERROR "Build the media engine with its FFmpeg and libass development headers; see third_party/media-engine/build-android.sh")
endif()
add_library(ffmpegJNI SHARED "${PORTICO_MEDIA3_SOURCE}/src/main/jni/ffmpeg_jni.cc")
target_include_directories(ffmpegJNI PRIVATE "${PORTICO_MEDIA_ENGINE}/include")
target_link_libraries(ffmpegJNI PRIVATE avcodec avutil swresample log)
add_library(porticosubtitles SHARED portico_subtitles.c)
target_include_directories(porticosubtitles PRIVATE "${PORTICO_MEDIA_ENGINE}/include")
target_link_libraries(porticosubtitles PRIVATE ass avcodec avformat avutil android jnigraphics log)
# Imported dependencies are packaged by portico-mpv once, not duplicated by this module.

target_compile_options(porticosubtitles PRIVATE -Wall -Wextra -Werror -ffile-prefix-map=${CMAKE_SOURCE_DIR}=/portico-media-jni)
target_compile_options(ffmpegJNI PRIVATE -ffile-prefix-map=${PORTICO_MEDIA3_SOURCE}=/media3-ffmpeg)
